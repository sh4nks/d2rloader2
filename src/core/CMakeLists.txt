add_library(Core STATIC)

set_source_files_properties(
    Log.qml
    PROPERTIES
        QT_QML_SINGLETON_TYPE TRUE
)

ecm_add_qml_module(Core GENERATE_PLUGIN_SOURCE
    URI com.someblocks.d2rloader.core
    OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/src/com/someblocks/d2rloader/core
    QML_FILES
        Log.qml
        CardHeader.qml
    DEPENDENCIES
        QtCore
        QtQml
        QtQuick
        org.kde.kirigami
    SOURCES
        logbuffer.h
        logbuffer.cpp
        paths.h
        updatechecker.h
        updatechecker.cpp
)

target_sources(Core PRIVATE
    logging.h
    logging.cpp
    verifieddownload.h
    verifieddownload.cpp
)

kconfig_target_kcfg_file(Core
    FILE d2rloaderconfig.kcfg
    CLASS_NAME D2RLoaderConfig
    MUTATORS
    DEFAULT_VALUE_GETTERS
    NOTIFIERS
    GENERATE_PROPERTIES
    SINGLETON
    QML_REGISTRATION
    GENERATE_MOC
    SOURCE_INCLUDE_FILES logbuffer.h
)

# d2rloader-version.h, generated by ecm_setup_version() in the top level
target_include_directories(Core PRIVATE ${CMAKE_BINARY_DIR})
target_include_directories(Core PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(Core PUBLIC
    Qt::Core
    Qt::Qml
    Qt::Quick
    Qt::Network
    KF6::ConfigCore
    KF6::ConfigGui
    KF6::I18n
    KF6::Notifications
)
